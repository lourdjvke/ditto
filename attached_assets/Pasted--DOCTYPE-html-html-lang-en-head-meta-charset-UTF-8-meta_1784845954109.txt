    <script>
        // Data Structure
        const configData = {
            categories: ['MATERIAL', 'COLOR', 'SIZE'],
            options: {
                'MATERIAL': [
                    { id: 'mat-smooth', name: 'Smooth Finish', rx: 0, stroke: 'transparent', width: 180 },
                    { id: 'mat-rounded', name: 'Soft Fabric', rx: 40, stroke: '#e5e5e5', width: 180 },
                    { id: 'mat-circle', name: 'Metallic Orb', rx: 90, stroke: '#d1d1d1', width: 180 },
                    { id: 'mat-tall', name: 'Tower Form', rx: 20, stroke: 'transparent', width: 120 },
                    { id: 'mat-slim', name: 'Slim Profile', rx: 10, stroke: '#ccc', width: 80 }
                ],
                'COLOR': [
                    { id: 'col-natural', name: 'Natural', value: '#e8e8e8' },
                    { id: 'col-black', name: 'Black Anthracite', value: '#2c2c2c' },
                    { id: 'col-gold', name: 'Gold Tone', value: '#d4af37' },
                    { id: 'col-bronze', name: 'Bronze Tone', value: '#cd7f32' },
                    { id: 'col-brass', name: 'Brass Tone', value: '#b5a642' },
                    { id: 'col-chestnut', name: 'Chestnut Brown', value: '#5c4033' },
                    { id: 'col-bitter', name: 'Bitter Orange', value: '#d95030' },
                    { id: 'col-zesty', name: 'Zesty Orange', value: '#ff9900' },
                    { id: 'col-radiant', name: 'Radiant Red', value: '#cc0000' },
                    { id: 'col-lime', name: 'Lime Green', value: '#32cd32' },
                    { id: 'col-navy', name: 'Navy Blue', value: '#000080' }
                ],
                'SIZE': [
                    { id: 'size-xs', name: 'Extra Small', scale: 0.5 },
                    { id: 'size-s', name: 'Compact', scale: 0.7 },
                    { id: 'size-m', name: 'Standard', scale: 1.0 },
                    { id: 'size-l', name: 'Large', scale: 1.2 },
                    { id: 'size-xl', name: 'Grand', scale: 1.4 }
                ]
            }
        };

        // State
        let activeCategory = 'COLOR';
        let selections = {
            'MATERIAL': configData.options['MATERIAL'][1], // Default to Soft Fabric
            'COLOR': configData.options['COLOR'][0],       // Default to Natural
            'SIZE': configData.options['SIZE'][2]          // Default to Standard
        };

        // DOM Elements
        const tabsContainer = document.getElementById('category-tabs');
        const listContainer = document.getElementById('options-list');
        const optionsContainer = document.getElementById('options-container');
        const svgShape = document.getElementById('custom-shape');
        const svgRect = document.getElementById('svg-rect');

        // Render Tabs
        function renderTabs() {
            tabsContainer.innerHTML = '';
            configData.categories.forEach(cat => {
                const btn = document.createElement('button');
                const isActive = cat === activeCategory;
                
                btn.className = `uppercase tracking-[0.15em] text-[10px] md:text-xs font-medium pb-2 relative focus:outline-none transition-colors ${isActive ? 'text-black' : 'text-gray-400'}`;
                btn.innerText = cat;
                
                if (isActive) {
                    const underline = document.createElement('div');
                    underline.className = 'absolute bottom-0 left-0 w-full h-[1.5px] bg-black';
                    btn.appendChild(underline);
                }

                btn.onclick = () => {
                    activeCategory = cat;
                    renderTabs();
                    renderOptions();
                    
                    // Reset scroll to top when switching tabs
                    optionsContainer.scrollTo({
                        top: 0,
                        behavior: 'smooth'
                    });
                };
                tabsContainer.appendChild(btn);
            });
        }

        // Render Options List
        function renderOptions() {
            listContainer.innerHTML = '';
            const options = configData.options[activeCategory];
            const currentSelection = selections[activeCategory];

            options.forEach((opt, index) => {
                const li = document.createElement('li');
                const isActive = opt.id === currentSelection.id;
                
                li.className = `option-item relative cursor-pointer group pr-4 md:pr-6 ${isActive ? 'active' : ''}`;
                li.id = `opt-${opt.id}`;
                
                li.innerHTML = `
                    <span class="transition-colors duration-300 group-hover:text-black block pr-2">${opt.name}</span>
                    ${activeCategory === 'COLOR' && isActive ? `<div class="text-[9px] md:text-[10px] tracking-wider uppercase mt-1 text-gray-500 pr-2">Signature</div>` : ''}
                    <div class="option-active-indicator"></div>
                `;

                li.onclick = (e) => {
                    selections[activeCategory] = opt;
                    renderOptions();
                    updateVisual();
                };

                listContainer.appendChild(li);
            });
        }

        // Update SVG based on selections
        function updateVisual() {
            const colorOpt = selections['COLOR'];
            const materialOpt = selections['MATERIAL'];
            const sizeOpt = selections['SIZE'];

            // Update Color
            svgRect.setAttribute('fill', colorOpt.value);
            
            // Adjust stroke visibility
            const isDark = ['col-black', 'col-chestnut', 'col-navy'].includes(colorOpt.id);
            svgRect.setAttribute('stroke', isDark ? 'rgba(255,255,255,0.1)' : materialOpt.stroke);

            // Update Material shape properties
            svgRect.setAttribute('rx', materialOpt.rx);
            
            // Adjust width, keeping it centered on the 200x200 canvas
            const newWidth = materialOpt.width;
            const xOffset = (200 - newWidth) / 2;
            svgRect.setAttribute('width', newWidth);
            svgRect.setAttribute('x', xOffset);

            // Update Scale (Size)
            svgShape.style.transform = `scale(${sizeOpt.scale})`;
        }

        // Initialize App
        renderTabs();
        renderOptions();
        updateVisual();

    </script>
